\begin{frame}
\fmt{HMC - Setup}
\bi
\i Define vector $\rho := (\beta, \nu) \in \mathbb{R}^n$. We want to sample from:
\begin{equation}
\label{eq:target_distribution}
g_d(\rho)d\pi(\rho) \; \propto \; \exp\left[-\frac 1 \xi  f(d, \rho)\right]d\pi(\rho)
\end{equation}
\i By taking logs, define potential energy term:

$$
\mathcal{U}(\rho):= \frac 1 \xi  f(d, \rho) - \log d\pi(\rho) 
$$ 
\i Consider an auxiliary ``momentum'' variable  $\omega\in R^n$ where $\omega \sim \mathcal{N}(0, M)$. Then the system Hamiltonian is given by:
$$
\mathcal{H}(\rho, \omega) :=  \mathcal{U}(\rho) + \frac{1}{2} \omega'M^{-1}\omega
$$
\ei
\end{frame}

\begin{frame} \fmt{Hamiltonian Dynamics}
\bi
\i We can simulate the motion of a point in our sample space using Hamilton's equations:

\begin{equation}
\label{eq:hamiltonian_dynamics}
    \frac{d \rho}{d t} = \frac{\partial \mathcal{H}}{\partial \omega}, \quad \frac{d \omega}{d t}  = -\frac{\partial \mathcal{H}}{\partial \rho}
\end{equation}

\i Reversibility, invariance and volume preserving.
\i $\mathcal{H}$, ``energy function'',   defines  canonical distribution. %with probability density function, $$ P( \beta, \mu)=\frac 1 Z \exp(-U(\beta))\exp(-K(\mu))$$
with density
$$
P(\rho, \omega) = \frac{1}{Z} \exp\left(-\mathcal U(\rho)\right)\exp\left(-\frac{1}{2} \omega'M^{-1}\omega\right)
$$ 
\i Invariance $\Rightarrow$ dynamics moves in subset of constant prob.   \ei
\end{frame}

\begin{frame}
\fmt{HMC - The algorithm}
\vspace{-10pt}
\begin{enumerate}
    \item Initialize $\rho_{(0)}$ and sample momentum $\omega_{(0)} \sim N(0,M)$.

    \item Generate a state proposal $(\tilde{\rho}_{(0)}, \tilde \omega_{(0)})$ by evolving its position according to dynamic system in \eqref{eq:hamiltonian_dynamics}, using the leapfrog integrator with step size $\epsilon$ for $L$ steps.
\bi \i Preserves canonical probability\ei
    \item Perform a Metropolis test for the state update
          $(\rho_{(1)}, \omega_{(1)}) \leftarrow (\tilde{\rho}_{(0)}, \tilde \omega_{(0)})$, with the
          acceptance probability:
          \begin{equation*}
              \min \left\{1, \ \exp \left( \mathcal{H}(\rho_{(0)}, \omega_{(0)}) - \mathcal{H}(\tilde{\rho}_{(0)}, \tilde \omega_{(0)}) \right) \right\}
          \end{equation*}

    \item Repeat steps 2-4 until 1000 samples (plus 500 burn-in samples) are drawn.
\end{enumerate}\\[10pt]

Note: We use the Stan software implementation, which uses the NUTS algorithm, an adaptive variant of HMC.
\end{frame}
